## Installation

```bash
pnpm dlx shadcn@latest add https://scpcn.com/r/spinner.json
```

## Usage

```tsx
import { Spinner } from "@/components/ui/warcraftcn/spinner";

<Spinner />
```

## Size

```tsx
<Spinner className="size-6" />

<Spinner className="size-14" />
```

## In Buttons

