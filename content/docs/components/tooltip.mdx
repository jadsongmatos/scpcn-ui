---
title: Tooltip
description: A popup that displays SCP Foundation-styled information when the element receives keyboard focus or the mouse hovers over it.
---

import { Button } from "@/components/ui/scpcn/button";
import {
Tooltip,
TooltipBody,
TooltipContent,
TooltipTitle,
TooltipTrigger,
} from "@/components/ui/scpcn/tooltip";

<Tooltip>
<TooltipTrigger asChild>
<Button className="w-64 md:w-96">Hover</Button>
</TooltipTrigger>
<TooltipContent>
<TooltipTitle>SCP-500</TooltipTitle>
<TooltipBody>Object Class: Safe</TooltipBody>
</TooltipContent>
</Tooltip>

## Installation

```bash
pnpm dlx shadcn@latest add https://scpcn.com/r/tooltip.json
```

## Usage

```tsx
import {
Tooltip,
TooltipBody,
TooltipContent,
TooltipTitle,
TooltipTrigger,
} from "@/components/ui/scpcn/tooltip";
```

```tsx
<Tooltip>
<TooltipTrigger>Hover</TooltipTrigger>
<TooltipContent>
<TooltipTitle>Item Name</TooltipTitle>
<TooltipBody>Item description</TooltipBody>
</TooltipContent>
</Tooltip>
```
