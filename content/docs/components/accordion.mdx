## Installation

```bash
pnpm dlx shadcn@latest add https://scpcn.com/r/accordion.json
```

## Usage

```tsx
import {
Accordion,
AccordionContent,
AccordionItem,
AccordionTrigger,
} from "@/components/ui/warcraftcn/accordion";

<Accordion type="single" collapsible>
<AccordionItem value="item-1">
<AccordionTrigger>Foundation Orders</AccordionTrigger>
<AccordionContent>
Establish containment quickly, then reinforce perimeter protocols before the
anomaly can adapt.
</AccordionContent>
</AccordionItem>
</Accordion>;
```
