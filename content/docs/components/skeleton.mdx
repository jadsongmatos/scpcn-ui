## Installation

```bash
pnpm dlx shadcn@latest add https://scpcn.com/r/skeleton.json
```

## Usage

```tsx
import { Skeleton } from "@/components/ui/warcraftcn/skeleton";

<Skeleton className="h-12 w-full" />;
```

## Shapes

```tsx
// Default rectangular shape
<Skeleton className="h-12 w-full" />

// Circular shape (for avatars)
<Skeleton variant="circular" className="size-12" />
```

## Classifications

```tsx
// Keter - red/dark with containment breach icons
<Skeleton classification="keter" className="h-12 w-full" />

// Euclid - teal/green with uncertainty icons
<Skeleton classification="euclid" className="h-12 w-full" />

// Thaumiel - blue with Foundation shield icons
<Skeleton classification="thaumiel" className="h-12 w-full" />

// Apollyon - purple with existential threat icons
```
