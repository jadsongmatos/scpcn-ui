## Installation

```bash
pnpm dlx shadcn@latest add https://scpcn.com/r/toast.json
```

## Usage

```tsx
import { triggerScrollToast } from "@/components/ui/scpcn/toast";
import { Button } from "@/components/ui/scpcn/button";

export default function ToastExample() {
return (
<Button
variant="frame"
onClick={() => triggerScrollToast({ message: "This is test" })}
className="w-[200px] md:min-w-xs py-3 px-6"
>
Show Toast
</Button>
);
}
```
