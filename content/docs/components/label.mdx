## Installation

```bash
pnpm dlx shadcn@latest add https://scpcn.com/r/label.json
```

## Usage

```tsx
import { Label } from "@/components/ui/scpcn/label";

<Label htmlFor="email">Your Email</Label>
```

### With Required Indicator

```tsx
<Label htmlFor="name" required>Entity Designation</Label>
```

### Disabled State

```tsx
<Label htmlFor="locked" disabled>Locked Field</Label>
```
