## Installation

```bash
pnpm dlx shadcn@latest add https://scpcn.com/r/checkbox.json
```

## Usage

```tsx
import { Checkbox } from "@/components/ui/warcraftcn/checkbox";
```

```tsx
<Checkbox classification="safe" id="terms">
Accept terms and conditions
</Checkbox>
```

### Classification Variants

```tsx
<Checkbox classification="thaumiel">Thaumiel theme</Checkbox>
<Checkbox classification="keter">Keter theme</Checkbox>
<Checkbox classification="euclid">Euclid theme</Checkbox>
<Checkbox classification="apollyon">Apollyon theme</Checkbox>
```

### Disabled State

```tsx
<Checkbox classification="safe" disabled>
Locked setting
</Checkbox>
```
